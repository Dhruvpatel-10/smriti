# Configuration Reference

## config.toml

```toml
[database]
path = "data/"                          # SurrealKV storage (relative to project root)

[pruning]
working_scratch_ttl_hours = 48          # scratch notes expire after 48h
working_priority_ttl_days = 7           # priorities expire after 7d
working_open_question_ttl_days = 30     # open questions expire after 30d
access_decay_interval_days = 7          # legacy decay interval for pre-SM-2 memories
importance_archive_threshold = 1        # memories at or below this importance get archived
consolidation_interval_hours = 24       # how often the consolidation tick runs (auto + candidate scan)

[consolidation]
auto_enabled = false                    # enable server-side auto-consolidation loop (ADR-026); default off
min_age_days = 7                        # minimum source memory age (days) before cluster candidacy
min_confidence = 0.75                   # LLM confidence threshold to trigger a merge
max_per_tick = 5                        # max clusters merged per periodic tick
max_cluster_size = 12                   # reject clusters with more source memories than this
cooldown_days = 7                       # days before re-evaluating a previously-deferred cluster

[queue]
drain_interval_secs = 300               # drain session runs every 5 minutes
max_queue_size = 1000                   # max pending queue entries

[alerts]
overdue_action_item_days = 1            # days past event_at before action item is overdue
stale_contact_days = 14                 # days since last_contact before entity is stale
expiring_memory_hours = 24              # hours before expiry to flag for promotion

[export]
backup_dir = "archive/exports"          # where kb_export writes JSONL files
exclude_sensitive = true                # never export sensitive-visibility memories

[embeddings]
enabled = false                         # flip to true to enable vector search + dedup
provider = "local"                      # "local" (fastembed, default) or "voyage" (cloud, ADR-016)
model = "embeddinggemma-300m"           # ONNX model for local provider (EmbeddingGemma-300M)
dimensions = 256                        # MRL truncation for local provider (256 recommended)

[embeddings.voyage]
# Only required when provider = "voyage".
# api_key = "va-..."                    # or set VOYAGE_API_KEY env var
model = "voyage-context-3"              # default; 1024-dim Matryoshka, 32K token cap, $0.18/M
output_dimension = 1024                 # Matryoshka truncation: 256 | 512 | 1024 | 2048
accepted_terms = false                  # must be true; Voyage no-training guarantee requires paid tier (ADR-016)

[embeddings.voyage.reranker]
enabled = false                         # opt-in stage 3.5 cross-encoder rerank (ADR-017)
model = "rerank-2.5-lite"               # Voyage rerank model
top_k_input = 50                        # RRF pool size sent to reranker
policy = "adaptive"                     # RerankerPolicy: "always" | "adaptive" (default) | "never" (ADR-022)

[search.fusion]
strategy = "rrf"                        # "rrf" (default, k=60) | "convex_combination" (α·vec + (1-α)·bm25)
vector_weight = 0.7                     # α for convex combination; ignored when strategy = "rrf"
min_score = 0.0                         # convex combination: drop candidates below this normalised score

[search.spreading]
cap = 3                                 # max memories added by spreading activation (ADR-023)
hop_decay = 0.5                         # score multiplier per hop depth (0.5^hop_depth)

[search.spreading.edge_weights]
derives = 1.0                           # strongest — derived memories carry full causal weight
extends = 0.8                           # extension/elaboration
updates = 0.8                           # supersession chain
relates = 0.5                           # weak topical link
involves = 0.3                          # entity co-occurrence
typed_relation = 0.9                    # LLM-extracted edge; scaled by edge confidence at query time

[markdown]
enabled = true                          # auto-regenerate ~/.claude/CLAUDE.md
output_path = "~/.claude/CLAUDE.md"     # supports ~ expansion

[server.auth]
# Path to the TOML file containing per-user bearer tokens (ADR-027).
# Default: /etc/smriti/tokens.toml (mode 0640, root:smriti).
# If the file is missing or empty, auth is disabled with a startup warning.
tokens_file = "/etc/smriti/tokens.toml"
```

## Bearer Auth Tokens (ADR-027, HTTP daemon mode only)

The HTTP daemon enforces `Authorization: Bearer <token>` on every request.
Tokens are stored in `/etc/smriti/tokens.toml` (generated by `install-system.sh`):

```toml
[tokens]
stark     = "<openssl rand -hex 32 output>"
leanscale = "<openssl rand -hex 32 output>"
```

To register the MCP server with auth (per user, run once):

```sh
claude mcp add smriti --transport http http://127.0.0.1:47821/mcp \
  --header "Authorization: Bearer <your-token>" -s user
```

Your token is in `/etc/smriti/tokens.toml` (requires `smriti` group membership to
read: `sudo cat /etc/smriti/tokens.toml`).

To rotate tokens: edit the file and `systemctl restart smriti`.

## Environment Variables

```
KB_DATA_DIR                   — override database path
KB_QUEUE_DIR                  — override queue directory
KB_ARCHIVE_DIR                — override archive directory
GEMINI_API_KEY                — Google Gemini API key (optional, for drain LLM)
GROQ_API_KEY                  — Groq API key (optional, fallback LLM)
VOYAGE_API_KEY                — Voyage AI API key (required when provider = "voyage")
SMRITI_FORCE_REEMBED_ON_START — set to "1" to bypass embedder fingerprint mismatch at startup;
                                 operator must run kb_reembed immediately after; prevents
                                 systemd Restart=always crash loops on model swap
```

## MCP Server Connection

Add to `~/.claude/settings.json`:

```json
{
  "mcpServers": {
    "kb": {
      "command": "/home/stark/.cargo/bin/smriti",
      "args": [],
      "env": {
        "KB_DATA_DIR": "/home/stark/projects/smriti/data",
        "KB_QUEUE_DIR": "/home/stark/projects/smriti/queue",
        "KB_ARCHIVE_DIR": "/home/stark/projects/smriti/archive"
      }
    }
  }
}
```
